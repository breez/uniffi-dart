/// Native asset id of the generated callback relay library.
const uniffiCallbackRelayAsset = '@ASSET@';
@Native<Uint64 Function()>(assetId: uniffiCallbackRelayAsset, symbol: '@PROBE@')
external int _uniffiRelayAbiHash();
@Native<Uint64 Function(Int64, Pointer<Void>)>(
  assetId: uniffiCallbackRelayAsset,
  symbol: 'uniffi_dispatch_create',
)
external int _uniffiRelayCreate(int port, Pointer<Void> api);
@Native<Void Function(Uint64)>(
  assetId: uniffiCallbackRelayAsset,
  symbol: 'uniffi_dispatch_drain',
)
external void _uniffiRelayDrain(int endpoint);
@Native<Bool Function(Uint64)>(
  assetId: uniffiCallbackRelayAsset,
  symbol: 'uniffi_dispatch_pending',
)
external bool _uniffiRelayPending(int endpoint);
@Native<UniffiRustFutureContinuationCallback>(
  assetId: uniffiCallbackRelayAsset,
  symbol: 'uniffi_dispatch_continue',
)
external void _uniffiRelayContinue(int data, int pollResult);
@Native<Uint64 Function(Uint64)>(
  assetId: uniffiCallbackRelayAsset,
  symbol: 'uniffi_dispatch_handle_new',
)
external int _uniffiRelayNewHandle(int endpoint);
@Native<Void Function(Uint64)>(
  assetId: uniffiCallbackRelayAsset,
  symbol: 'uniffi_dispatch_handle_release',
)
external void _uniffiRelayReleaseHandle(int handle);

/// A callback handle from the relay, which records that this isolate owns it
/// and sends every call on it here. Null when the relay isn't bundled.
int? _uniffiRelayHandleNew() {
  final endpoint = UniffiCallbackRelay.endpoint;
  return endpoint == null ? null : _uniffiRelayNewHandle(endpoint);
}

/// Tells the relay that Rust has given [handle] up.
void _uniffiRelayHandleRelease(int handle) {
  if (UniffiCallbackRelay._endpoint != null) _uniffiRelayReleaseHandle(handle);
}

/// Whether the relay has calls queued for this isolate. The relay runs at
/// most 64 per wake-up, so the barrier checks this before completing.
bool _uniffiRelayHasQueuedCalls() {
  final id = UniffiCallbackRelay._checked ? UniffiCallbackRelay._endpoint : null;
  return id != null && _uniffiRelayPending(id);
}
const _uniffiExpectedRelayAbiHash = @HASH@;

/// This isolate's connection to the generated callback relay library.
///
/// When the library is bundled, every callback goes through it: inline on this
/// isolate's thread, and queued to this isolate from any other thread. Without
/// it, async methods and frees still arrive through listeners, but synchronous
/// callbacks can only run on this isolate's thread.
abstract final class UniffiCallbackRelay {
  static bool _checked = false;
  static int? _endpoint;
  static RawReceivePort? _port;
  static final _warned = <String>{};

  /// The Rust futures this isolate is polling through the relay, by id.
  static final _continuations = <int, void Function(int)>{};
  static int _lastContinuation = 0;

  /// This isolate's relay endpoint, or null when the library isn't bundled.
  static int? get endpoint {
    if (_checked) return _endpoint;
    _checked = true;
    final int hash;
    try {
      hash = _uniffiRelayAbiHash();
    } catch (_) {
      // The asset isn't bundled.
      return null;
    }
    if (hash != _uniffiExpectedRelayAbiHash) {
      throw StateError(
        'The callback relay library does not match these bindings (its ABI '
        'hash is 0x${hash.toRadixString(16)}, the bindings expect '
        '0x${_uniffiExpectedRelayAbiHash.toRadixString(16)}). Rebuild the '
        'uniffi_dispatch crate generated with these bindings.',
      );
    }
    // The relay wakes this isolate by posting to the port: null to run queued
    // callbacks, or a Rust future's wake-up as an integer. Posting to a closed
    // port fails instead of aborting, so the relay notices when the isolate
    // has exited. Like listeners, the port doesn't keep the isolate alive,
    // except while a Rust future is pending.
    final port = RawReceivePort((Object? message) {
      if (message is int) {
        // The future's id, then the poll result in the low byte.
        _continuations[message >> 8]?.call((message & 0xff).toSigned(8));
        return;
      }
      final id = _endpoint;
      if (id != null) _uniffiRelayDrain(id);
    }, 'uniffi callback relay');
    port.keepIsolateAlive = false;
    final id = _uniffiRelayCreate(
      port.sendPort.nativePort,
      NativeApi.initializeApiDLData,
    );
    if (id == 0) {
      port.close();
      throw UnsupportedError(
        'This Dart VM does not provide the isolate checks the callback relay '
        'needs',
      );
    }
    _port = port;
    _endpoint = id;
    return id;
  }

  /// Prints [message] once per [component] in this isolate.
  static void warnMissing(String component, String message) {
    if (_warned.add(component)) print(message);
  }
}

/// A Rust future's continuation through the relay. Rust may call it from any
/// thread, even after this isolate has exited: the relay posts each wake-up to
/// this isolate's port, and drops it if the isolate has gone. While any are
/// open, the port keeps the isolate alive, as a listener would.
final class _UniffiRelayContinuation implements _UniffiContinuation {
  _UniffiRelayContinuation._(this.data, this._id);

  static final _continue =
      Native.addressOf<NativeFunction<UniffiRustFutureContinuationCallback>>(
        _uniffiRelayContinue,
      );

  static _UniffiRelayContinuation? open(void Function(int) onPoll) {
    final endpoint = UniffiCallbackRelay.endpoint;
    // The endpoint shares the callback data with the id.
    if (endpoint == null || endpoint >= 1 << 32) return null;
    final continuations = UniffiCallbackRelay._continuations;
    var id = UniffiCallbackRelay._lastContinuation;
    do {
      id = (id + 1) & 0xffffffff;
    } while (continuations.containsKey(id));
    UniffiCallbackRelay._lastContinuation = id;
    continuations[id] = onPoll;
    UniffiCallbackRelay._port!.keepIsolateAlive = true;
    return _UniffiRelayContinuation._((endpoint << 32) | id, id);
  }

  @override
  final int data;
  final int _id;

  @override
  Pointer<NativeFunction<UniffiRustFutureContinuationCallback>> get callback =>
      _continue;

  @override
  void close() {
    final continuations = UniffiCallbackRelay._continuations;
    continuations.remove(_id);
    if (continuations.isEmpty) {
      UniffiCallbackRelay._port!.keepIsolateAlive = false;
    }
  }
}

_UniffiContinuation? _uniffiRelayContinuation(void Function(int) onPoll) =>
    _UniffiRelayContinuation.open(onPoll);

/// The relay's NativeFinalizer for one object type, or null when the relay
/// isn't bundled. [register] first gives the relay the type's UniFFI free.
NativeFinalizer? uniffiRelayObjectFinalizer(
  void Function() register,
  Pointer<NativeFinalizerFunction> Function() finalize,
) {
  if (UniffiCallbackRelay.endpoint == null) return null;
  register();
  return NativeFinalizer(finalize());
}
